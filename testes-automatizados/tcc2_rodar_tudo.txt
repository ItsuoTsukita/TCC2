Coordena a bateria inicial de cargas e guarda os resultados na AWS.

#!/usr/bin/env bash
set -Eeuo pipefail

REGIAO=us-east-1
BUCKET=tcc2-pacotes-CONTA_AWS-us-east-1
P1=i-EXEMPLO
P2=i-EXEMPLO
MOTOR=i-EXEMPLO
IDS=("$P1" "$P2" "$MOTOR")
VOLUMES=(10000 100000 500000)
MOTORES=(risingwave flink arroyo kinesis)
RUN="$(date -u +%Y%m%d_%H%M%S)"
PREFIXO="resultados/automatico/$RUN"
LIMITE=$(( $(date +%s) + 5400 ))
STREAM="tcc2-auto-$RUN"
STREAM_CRIADO=0
STREAMS_CRIADOS=()
LIGOU=0
ETAPA=preflight

mensagem() { printf '\n[%s] %s\n' "$(date -u +%FT%TZ)" "$*"; }
falhar() { mensagem "FALHA: $*" >&2; return 1; }
verificar_tempo() {
  (( $(date +%s) < LIMITE )) || falhar 'Limite de 90 minutos atingido; interrupção preventiva.'
}

parar_tudo() {
  local codigo=$?
  trap - EXIT INT TERM
  mensagem 'FINALIZAÇÃO: interrompendo somente as três EC2 utilizadas por este teste.'
  if (( LIGOU )); then
    aws ec2 stop-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null 2>&1 || true
    if aws ec2 wait instance-stopped --region "$REGIAO" --instance-ids "${IDS[@]}"; then
      mensagem 'EC2-01, EC2-02 e EC2-05 desligadas.'
    else
      mensagem 'ATENÇÃO: confirme manualmente o estado das instâncias.'
      codigo=1
    fi
  fi
  for stream_de_exclusao in "${STREAMS_CRIADOS[@]}"; do
    mensagem "Excluindo SOMENTE o stream Kinesis criado nesta execução: $stream_de_exclusao"
    local estado_stream
    estado_stream=$(aws kinesis describe-stream-summary --region "$REGIAO" \
      --stream-name "$stream_de_exclusao" --query StreamDescriptionSummary.StreamStatus \
      --output text 2>/dev/null || true)
    if [[ "$estado_stream" != '' && "$estado_stream" != DELETING ]]; then
      aws kinesis delete-stream --region "$REGIAO" --stream-name "$stream_de_exclusao" \
        --enforce-consumer-deletion >/dev/null 2>&1 || { mensagem 'ATENÇÃO: stream não pôde ser excluído!'; codigo=1; }
    fi
  done
  mensagem "Resultados disponíveis em s3://$BUCKET/$PREFIXO/"
  mensagem "Última etapa: $ETAPA; código final: $codigo"
  exit "$codigo"
}
trap parar_tudo EXIT INT TERM

for ferramenta in aws jq python3; do command -v "$ferramenta" >/dev/null || falhar "Falta $ferramenta"; done
aws sts get-caller-identity --query Account --output text | grep -qx CONTA_AWS || falhar 'Conta AWS inesperada.'

if [[ "${1:-}" == '--ajuda' ]]; then
  echo 'Uso: bash tcc2_rodar_tudo.sh [--apenas risingwave|flink|arroyo|kinesis]'
  exit 0
fi
if [[ "${1:-}" == '--apenas' ]]; then
  case "${2:-}" in risingwave|flink|arroyo|kinesis) MOTORES=("$2");; *) falhar 'Motor inválido';; esac
elif (( $# != 0 )); then falhar 'Argumento desconhecido'; fi

mensagem 'PREFLIGHT: conferindo EC2, S3, instâncias e custo antes de iniciar.'
for id in "${IDS[@]}"; do
  estado=$(aws ec2 describe-instances --region "$REGIAO" --instance-ids "$id" \
    --query 'Reservations[0].Instances[0].State.Name' --output text)
  mensagem "$id: $estado"
  [[ "$estado" == stopped ]] || falhar "É preciso que $id esteja stopped. Nada iniciado."
done
for nome in Pacote_Envio_AWS_20261106_000040.zip TCC2_Complementos_20261105_221657.zip TCC2_Automacao_Corrigido_20261105_210633.zip; do
  aws s3api head-object --region "$REGIAO" --bucket "$BUCKET" --key "pacotes/$nome" >/dev/null || falhar "Pacote ausente: $nome"
done
mensagem "Preflight concluído. Serão ligadas 3 EC2 c7a.2xlarge; Materialize NÃO será repetido."

enviar() {
  local id="$1" script="$2" parametros
  parametros=$(jq -nc --arg cmd "$script" '{commands:[$cmd],executionTimeout:["3600"]}')
  aws ssm send-command --region "$REGIAO" --document-name AWS-RunShellScript \
    --instance-ids "$id" --parameters "$parametros" \
    --query 'Command.CommandId' --output text
}

aguardar() {
  local id="$1" comando="$2" estado='' cont=0
  while (( cont++ < 900 )); do
    verificar_tempo || return 1
    estado=$(aws ssm get-command-invocation --region "$REGIAO" \
      --instance-id "$id" --command-id "$comando" --query Status \
      --output text 2>/dev/null || true)
    case "$estado" in Success|Failed|TimedOut|Cancelled|DeliveryTimedOut|Undeliverable|Terminated) break;; esac
    sleep 5
  done
  aws ssm get-command-invocation --region "$REGIAO" --command-id "$comando" \
    --instance-id "$id" \
    --query '{Estado:Status,Saida:StandardOutputContent,Erro:StandardErrorContent}' \
    --output json || true
  [[ "$estado" == Success ]]
}

rodar() { local id="$1" script="$2" cmd; cmd=$(enviar "$id" "$script"); aguardar "$id" "$cmd"; }
injetar() {
  local s="$1" chave="$2" valor="$3"
  s="${s//__${chave}__/$valor}"
  printf '%s' "$s"
}

mensagem 'LIGANDO EC2-01, EC2-02 e EC2-05.'
aws ec2 start-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null
LIGOU=1
aws ec2 wait instance-running --region "$REGIAO" --instance-ids "${IDS[@]}"

online=0
for tentativa in $(seq 1 45); do
  ativos=" $(aws ssm describe-instance-information --region "$REGIAO" \
    --query 'InstanceInformationList[?PingStatus==`Online`].InstanceId' --output text | tr '\t' ' ') "
  online=1
  for id in "${IDS[@]}"; do [[ "$ativos" == *" $id "* ]] || online=0; done
  (( online )) && break
  sleep 10
done
(( online )) || falhar 'SSM offline. EC2 serão desligadas.'

for id in "${IDS[@]}"; do
  rodar "$id" $'#!/bin/bash\nset -e\nshutdown -h +50\necho PARADA_DE_SEGURANCA_EM_50_MINUTOS' || falhar "Agendamento de proteção falhou: $id"
done

CREATE TABLE eventos_transporte (
  id_evento BIGINT, veiculo_id BIGINT, latitude DOUBLE, longitude DOUBLE,
  velocidade DOUBLE, temperatura_carga DOUBLE, status_rota STRING, data_hora STRING,
  evento_ts AS TO_TIMESTAMP(REPLACE(data_hora, 'T', ' ')),
  WATERMARK FOR evento_ts AS evento_ts - INTERVAL '5' SECOND
) WITH ('connector'='kafka', 'topic'='eventos-transporte',
        'properties.bootstrap.servers'='kafka:29092',
        'properties.group.id'='tcc2-auto-flink',
        'scan.startup.mode'='earliest-offset','format'='json');
CREATE TABLE resumo_frota_sink (
 status_rota STRING,total_eventos BIGINT,velocidade_media DOUBLE,
 temperatura_media DOUBLE, PRIMARY KEY(status_rota) NOT ENFORCED
) WITH ('connector'='jdbc','url'='jdbc:postgresql://postgres:5432/tcc',
        'table-name'='resumo_frota','username'='tcc','password'='__SENHA__');
CREATE TABLE posicao_atual_sink (
 veiculo_id BIGINT,latitude DOUBLE,longitude DOUBLE,velocidade DOUBLE,
 temperatura_carga DOUBLE,status_rota STRING,data_hora TIMESTAMP(3),
 PRIMARY KEY(veiculo_id) NOT ENFORCED
) WITH ('connector'='jdbc','url'='jdbc:postgresql://postgres:5432/tcc',
        'table-name'='posicao_atual','username'='tcc','password'='__SENHA__');
EXECUTE STATEMENT SET BEGIN
INSERT INTO resumo_frota_sink
 SELECT status_rota,COUNT(*),AVG(velocidade),AVG(temperatura_carga)

INSERT INTO posicao_atual_sink
 SELECT veiculo_id,latitude,longitude,velocidade,temperatura_carga,status_rota,evento_ts
 FROM (SELECT *,ROW_NUMBER() OVER (PARTITION BY veiculo_id ORDER BY evento_ts DESC) AS rn
       FROM eventos_transporte) WHERE rn=1;
END;
SQL

      -c 'SELECT COUNT(*),COUNT(DISTINCT id_evento) FROM eventos_transporte;' | tail -1)
    ;;
  flink|arroyo)
    LINHA=$(docker exec kafka-tcc /opt/kafka/bin/kafka-get-offsets.sh \
      --bootstrap-server kafka:29092 --topic eventos-transporte --time -1 \
      | awk -F: '{s+=$3} END{print s+0}')
    if [ "$ENGINE" = flink ]; then
      for tentativa in $(seq 1 45); do
        SINK=$(docker exec postgres-resultados-tcc psql -U tcc -d tcc -t -A \
          -c 'SELECT COALESCE(SUM(total_eventos),0) FROM resumo_frota;')
        [[ "$SINK" == "$LINHA" ]] && break
        sleep 4
      done
      echo "FLINK_SINK_TOTAL=$SINK"
      export SINK
    fi
    ;;
  kinesis)
    for tentativa in $(seq 1 45); do
      LINHA=$(BASE="$BASE" /opt/tcc2/venv/bin/python - <<'PY'
import os,sqlite3
db=sqlite3.connect(os.path.join(os.environ['BASE'],'resultados_kinesis.sqlite'))
print(db.execute('SELECT COUNT(*) FROM eventos').fetchone()[0])
PY
      )
      [[ "$LINHA" == "$VOLUME" ]] && break
      sleep 4
    done
    ;;
esac
export LINHA
/opt/tcc2/venv/bin/python - <<'PY'
import os,json,boto3
from pathlib import Path
p=Path(os.environ['BASE']); engine=os.environ['ENGINE']; esperado=int(os.environ['VOLUME'])
linha=os.environ['LINHA'].strip().splitlines()[-1]
if engine=='risingwave':
    recebidos,distintos=map(int,linha.split('|'))
    classe='processamento_em_tabela'
elif engine=='flink':
    anterior=Path('/tmp/tcc2-auto-'+os.environ['RUN']+'-flink-anterior.json')
    total_kafka=int(linha); total_sink=int(os.environ['SINK'])
    prev=json.loads(anterior.read_text()) if anterior.exists() else {'kafka':0,'sink':0}
    recebidos=total_kafka-prev['kafka']; distintos=recebidos
    processados=total_sink-prev['sink']
    classe='ingestao_no_kafka_e_sink_agregado'
elif engine=='arroyo':
    anterior=Path('/tmp/tcc2-auto-'+os.environ['RUN']+'-arroyo-anterior.json')
    total_kafka=int(linha)
    prev=json.loads(anterior.read_text()) if anterior.exists() else {'kafka':0}
    recebidos=total_kafka-prev['kafka']; distintos=recebidos
    classe='ingestao_no_kafka_apenas'
else:
    recebidos=distintos=int(linha); classe='eventos_unicos_no_consumidor_sqlite'
info=dict(engine=engine,volume=esperado,registros=recebidos,
          distintos=distintos,diferenca=esperado-recebidos,
          tipo_auditoria=classe)
if engine=='flink':
    info['processados_no_sink_por_diferenca']=processados
    info['diferenca_sink']=esperado-processados
if engine in ('flink','arroyo'):
    anterior.write_text(json.dumps({'kafka':total_kafka,'sink':int(os.environ['SINK']) if engine=='flink' else 0}))
(p/'consistencia.json').write_text(json.dumps(info,indent=2))
s3=boto3.client('s3',region_name='us-east-1')
prefix=f"{os.environ['PREFIXO']}/{engine}/{esperado}/motor/"
for nome in ('metricas.csv','consistencia.json','consumidor.log'):
    if (p/nome).exists():
        s3.upload_file(str(p/nome),'tcc2-pacotes-CONTA_AWS-us-east-1',prefix+nome,
                       ExtraArgs={'ServerSideEncryption':'AES256'})
print('AUDITORIA:',info,flush=True)
assert recebidos==esperado, 'Divergência de ingestão detectada; resultados salvos no S3'
if engine=='flink':
    assert processados==esperado, 'Divergência Flink/Postgres detectada; resultados salvos no S3'
PY
REMOTO
)

for engine in "${MOTORES[@]}"; do
  verificar_tempo
  ETAPA="$engine:preparacao"
  mensagem "========== $engine =========="
  case "$engine" in
    risingwave) rodar "$MOTOR" "$SQL_RISING" || falhar 'RisingWave não inicializou';;
    flink) rodar "$MOTOR" "$INICIAR_FLINK" || falhar 'Flink/Kafka não inicializou';;
    arroyo) rodar "$MOTOR" "$INICIAR_ARROYO" || falhar 'Arroyo/Kafka não inicializou';;
    kinesis) :;;
  esac
  for volume in "${VOLUMES[@]}"; do
    verificar_tempo
    ETAPA="$engine:$volume"
    mensagem "--- $engine / $volume eventos, 2 produtores paralelos ---"
    for id in "${IDS[@]}"; do
      rodar "$id" $'#!/bin/bash\nset -e\nshutdown -c >/dev/null 2>&1 || true\nshutdown -h +45\necho PARADA_DE_SEGURANCA_REAGENDADA' || falhar 'Não foi possível renovar desligamento seguro.'
    done
    case "$engine" in
      risingwave) rodar "$MOTOR" "$PREPARAR_RISING" || falhar 'Tabela RisingWave';;
      flink)
        if [[ "$volume" == 10000 ]]; then
          rodar "$MOTOR" "$PREPARAR_FLINK" || falhar 'Job Flink inválido'
        fi
        ;;
      arroyo) :;;
      kinesis)
        STREAM="tcc2-auto-$RUN-$volume"
        aws kinesis create-stream --region "$REGIAO" --stream-name "$STREAM" --shard-count 2
        STREAMS_CRIADOS+=("$STREAM")
        for i in $(seq 1 90); do
          estado=$(aws kinesis describe-stream-summary --region "$REGIAO" --stream-name "$STREAM" \
            --query StreamDescriptionSummary.StreamStatus --output text)
          [[ "$estado" == ACTIVE ]] && break
          sleep 5
        done
        [[ "$estado" == ACTIVE ]] || falhar 'Stream Kinesis não ficou ACTIVE.'
        rodar "$MOTOR" "$(injetar "$(injetar "$(injetar "$PREPARAR_KINESIS" STREAM "$STREAM")" VOLUME "$volume")" RUN "$RUN")" || falhar 'Consumidor Kinesis'
        ;;
    esac
    monitor="$(injetar "$(injetar "$(injetar "$MONITOR" MOTOR "$engine")" VOLUME "$volume")" RUN "$RUN")"
    rodar "$MOTOR" "$monitor" || falhar 'Monitor de CPU/RAM'
    declare -a JOBS=()
    for num in 01 02; do
      if [[ "$num" == 01 ]]; then id="$P1"; else id="$P2"; fi
      script="$TESTAR"
      for par in "MOTOR=$engine" "VOLUME=$volume" "NUM=$num" "RUN=$RUN" "PREFIXO=$PREFIXO" "STREAM=$STREAM"; do
        script="$(injetar "$script" "${par%%=*}" "${par#*=}")"
      done
      JOBS+=("$id:$(enviar "$id" "$script")")
    done
    erros=0
    for job in "${JOBS[@]}"; do
      id="${job%%:*}"; cmd="${job#*:}"
      aguardar "$id" "$cmd" || erros=1
    done
    if [[ "$engine" == kinesis ]]; then sleep 10; else sleep 5; fi
    if [[ "$engine" == flink || "$engine" == arroyo ]]; then
      mensagem 'Atenção: Kafka é cumulativo nesta etapa; a auditoria dos eventos processados não é por ID.'
    fi
    validar="$VALIDAR"
    for par in "MOTOR=$engine" "VOLUME=$volume" "RUN=$RUN" "PREFIXO=$PREFIXO"; do
      validar="$(injetar "$validar" "${par%%=*}" "${par#*=}")"
    done
    rodar "$MOTOR" "$validar" || erros=1
    (( erros == 0 )) || falhar "$engine $volume apresentou erro, consulte S3. Outras etapas canceladas para poupar custos."
    mensagem "$engine $volume: JMeter e auditoria salvos."
  done
  mensagem "MOTOR $engine CONCLUÍDO"
done
