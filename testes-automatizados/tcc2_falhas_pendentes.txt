Executa os testes complementares de falha do Arroyo e do consumidor Kinesis.

#!/usr/bin/env bash
set -Eeuo pipefail
export AWS_PAGER=''
REGIAO=us-east-1
BUCKET=tcc2-pacotes-CONTA_AWS-us-east-1
P1=i-EXEMPLO
P2=i-EXEMPLO
MOTOR=i-EXEMPLO
IDS=("$P1" "$P2" "$MOTOR")
RUN=$(date -u +%Y%m%d_%H%M%S)
PREFIXO="resultados/falhas/streams/$RUN"
TOPICO="eventos-transporte-falhas-$RUN-arroyo"
SAIDA="resultados-arroyo-falhas-$RUN"
STREAM="tcc2-auto-$RUN-10000"
TEMPO_FINAL=$(( $(date +%s) + 3300 ))
LIGOU=0
STREAM_CRIADO=0
INCIDENTES=0
log(){ printf '[%s] %s\n' "$(date -u +%FT%TZ)" "$*"; }
encerrar(){
  local codigo=$?
  trap - EXIT INT TERM
  if (( LIGOU )); then
    log 'Desligando somente EC2-01, EC2-02 e EC2-05.'
    aws ec2 stop-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null 2>&1 || true
    if aws ec2 wait instance-stopped --region "$REGIAO" --instance-ids "${IDS[@]}"; then
      log 'Tres EC2 confirmadas desligadas.'
    else
      log 'ATENCAO: confirme o desligamento das 3 EC2 no console.'
      codigo=1
    fi
  fi
  if (( STREAM_CRIADO )); then
    log "Excluindo exclusivamente o stream temporario $STREAM"
    aws kinesis delete-stream --region "$REGIAO" --stream-name "$STREAM" \
      --enforce-consumer-deletion >/dev/null 2>&1 || { log 'ATENCAO: exclusao do stream falhou.'; codigo=1; }
  fi
  log "Evidencias: s3://$BUCKET/$PREFIXO/"
  log "Codigo final: $codigo; incidentes de execucao: $INCIDENTES"
  exit "$codigo"
}
trap encerrar EXIT INT TERM
for exe in aws jq; do command -v "$exe" >/dev/null || { log "Comando ausente: $exe"; exit 1; }; done
test "$(aws sts get-caller-identity --query Account --output text)" = CONTA_AWS
for id in "${IDS[@]}"; do
  estado=$(aws ec2 describe-instances --region "$REGIAO" --instance-ids "$id" \
    --query 'Reservations[0].Instances[0].State.Name' --output text)
  log "$id: $estado"
  test "$estado" = stopped || { log 'ABORTADO: EC2 nao estava desligada.'; exit 1; }
done
enviar(){
  local id="$1" script="$2" payload
  payload=$(jq -nc --arg cmd "$script" '{commands:[$cmd],executionTimeout:["1200"]}')
  aws ssm send-command --region "$REGIAO" --document-name AWS-RunShellScript \
    --instance-ids "$id" --parameters "$payload" --query Command.CommandId --output text
}
aguardar(){
  local id="$1" comando="$2" estado=''
  for _ in $(seq 1 180); do
    (( $(date +%s) < TEMPO_FINAL )) || { log 'Limite seguro de tempo atingido.'; return 1; }
    estado=$(aws ssm get-command-invocation --region "$REGIAO" --instance-id "$id" \
      --command-id "$comando" --query Status --output text 2>/dev/null || true)
    case "$estado" in Success|Failed|Cancelled|TimedOut|DeliveryTimedOut|Undeliverable) break;; esac
    sleep 5
  done
  aws ssm get-command-invocation --region "$REGIAO" --instance-id "$id" \
    --command-id "$comando" \
    --query '{Estado:Status,Saida:StandardOutputContent,Erro:StandardErrorContent}' --output json || true
  [[ "$estado" = Success ]]
}
rodar(){ local id="$1" cmd; cmd=$(enviar "$id" "$2"); aguardar "$id" "$cmd"; }
substituir(){ local s="$1" chave="$2" valor="$3"; printf '%s' "${s//__${chave}__/$valor}"; }
parametrizar(){
  local texto="$1" item
  for item in "ENGINE=$2" "RUN=$RUN" "TOPICO=$TOPICO" "SAIDA=$SAIDA" "STREAM=$STREAM" "PREFIXO=$PREFIXO" "NUM=${3:-01}"; do
    texto=$(substituir "$texto" "${item%%=*}" "${item#*=}")
  done
  printf '%s' "$texto"
}

log 'Iniciando EC2-01, EC2-02 e EC2-05 para testar SOMENTE Arroyo e Kinesis.'
aws ec2 start-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null
LIGOU=1
aws ec2 wait instance-running --region "$REGIAO" --instance-ids "${IDS[@]}"
online=0
for _ in $(seq 1 45); do
  ativos=" $(aws ssm describe-instance-information --region "$REGIAO" \
    --query 'InstanceInformationList[?PingStatus==`Online`].InstanceId' \
    --output text | tr '\t' ' ') "
  online=1
  for id in "${IDS[@]}"; do [[ "$ativos" = *" $id "* ]] || online=0; done
  (( online )) && break
  sleep 10
done
(( online )) || { log 'SSM offline; abortando.'; exit 1; }
for id in "${IDS[@]}"; do
  rodar "$id" $'#!/bin/bash\nset -e\nshutdown -h +55\necho PARADA_DE_SEGURANCA_55_MIN' || exit 1
done
