Executa o teste final de checkpoint do Flink, reinicia o TaskManager e confere a recuperação.

#!/usr/bin/env bash
set -Eeuo pipefail
export AWS_PAGER=''
REGIAO=us-east-1
BUCKET=tcc2-pacotes-CONTA_AWS-us-east-1
P1=i-EXEMPLO
P2=i-EXEMPLO
MOTOR=i-EXEMPLO
IPS_MOTOR=IP_EXEMPLO
IDS=("$P1" "$P2" "$MOTOR")
RUN=$(date -u +%Y%m%d_%H%M%S)
PREFIXO="resultados/checkpoints/flink/$RUN"
LIGOU=0
INCIDENTES=0
LIMITE=$(( $(date +%s) + 1500 ))
log(){ printf '[%s] %s\n' "$(date -u +%FT%TZ)" "$*"; }
finalizar(){
  local rc=$?
  trap - EXIT INT TERM
  if (( LIGOU )); then
    log 'Desligando somente EC2-01, EC2-02 e EC2-05.'
    aws ec2 stop-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null || rc=1
    if aws ec2 wait instance-stopped --region "$REGIAO" --instance-ids "${IDS[@]}"; then
      log 'Tres EC2 confirmadas desligadas.'
    else
      log 'ATENCAO: verifique as tres EC2 no console.'
      rc=1
    fi
    log "Evidencias: s3://$BUCKET/$PREFIXO/"
  fi
  log "Codigo final: $rc; incidentes: $INCIDENTES"
  exit "$rc"
}
trap finalizar EXIT INT TERM
modo="${1:---check}"
[[ "$modo" == --check || "$modo" == --run ]] || { log 'Use --check ou --run'; exit 2; }
for exe in aws jq python3; do
  command -v "$exe" >/dev/null || { log "Ausente: $exe"; exit 1; }
done
[[ "$(aws sts get-caller-identity --query Account --output text)" == CONTA_AWS ]] || {
  log 'Conta AWS divergente. Nada iniciado.'; exit 1;
}
dados=$(aws ec2 describe-instances --region "$REGIAO" --instance-ids "${IDS[@]}" --output json)
grupos=$(jq -r '.Reservations[].Instances[].SecurityGroups[].GroupId' <<<"$dados" | sort -u)
[[ -n "$grupos" ]] || { log 'SG nao localizado'; exit 1; }
readarray -t grupos_ids <<<"$grupos"
regras=$(aws ec2 describe-security-groups --region "$REGIAO" --group-ids "${grupos_ids[@]}" --output json)
python3 - "$dados" "$regras" "$P1" "$P2" "$MOTOR" <<'PY'
import ipaddress,json,sys
infos=json.loads(sys.argv[1]); regras=json.loads(sys.argv[2])
p1,p2,motor=sys.argv[3:]
maquinas={i['InstanceId']:i for r in infos['Reservations'] for i in r['Instances']}
assert set(maquinas)=={p1,p2,motor},'Instancias inesperadas'
sgs={g['GroupId']:g for g in regras['SecurityGroups']}
for id,inst in maquinas.items():
    print('EC2:',id,inst['State']['Name'],inst['InstanceType'],flush=True)
    assert inst['State']['Name']=='stopped','EC2 nao esta desligada'
    assert inst['InstanceType']=='c7a.2xlarge','Tipo EC2 inesperado'
    assert inst['VpcId']=='vpc-EXEMPLO','VPC divergente'
    assert inst['IamInstanceProfile']['Arn'].endswith('/TCC2-EC2-Kinesis'),'IAM divergente'
def regras_perm(a,b,saida):
    origem=maquinas[a]; destino=maquinas[b]
    if saida: lado,peer,field=origem,destino,'IpPermissionsEgress'
    else: lado,peer,field=destino,origem,'IpPermissions'
    ip=ipaddress.ip_address(peer['PrivateIpAddress'])
    sgr={g['GroupId'] for g in peer['SecurityGroups']}
    for g in lado['SecurityGroups']:
        for p in sgs[g['GroupId']].get(field,[]):
            proto=str(p.get('IpProtocol',''))
            if not (proto=='-1' or proto=='tcp' and p.get('FromPort',0)<=9092<=p.get('ToPort',0)):
                continue
            if any(x.get('GroupId') in sgr for x in p.get('UserIdGroupPairs',[])):
                return True
            if any(ip in ipaddress.ip_network(x['CidrIp'],strict=False) for x in p.get('IpRanges',[])):
                return True
    return False
for produtor in (p1,p2):
    assert regras_perm(produtor,motor,True),'Saida Kafka bloqueada: '+produtor
    assert regras_perm(produtor,motor,False),'Entrada Kafka bloqueada: '+produtor
print('PRECHECK_EC2_E_KAFKA_OK',flush=True)
PY
aws s3api head-bucket --bucket "$BUCKET" >/dev/null
for id in "${IDS[@]}"; do
  pendente=$(aws ssm list-command-invocations --region "$REGIAO" --instance-id "$id" --query "CommandInvocations[?Status=='Pending'||Status=='InProgress'||Status=='Delayed'].CommandId" --output text)
  [[ -z "$pendente" ]] || { log "Comandos SSM pendentes em $id"; exit 1; }
done
log 'PRECHECK OK: instancias desligadas, rede e S3 conferidos.'
if [[ "$modo" == --check ]]; then
  log 'Nenhuma EC2 iniciada. --run exige autorizacao.'
  exit 0
fi
log 'O teste liga TRES EC2 c7a.2xlarge por ate 25 minutos e gera custos.'
log 'Uma unica carga de 10.000 eventos, sem alterar os baselines nem chamar Kinesis.'
read -r -p 'Autorizar os custos? Digite SIM: ' confirmar
[[ "$confirmar" == SIM ]] || { log 'Cancelado sem iniciar EC2.'; exit 0; }
enviar(){
  local id="$1" script="$2" payload
  payload=$(jq -cn --arg s "$script" '{commands:[$s],executionTimeout:["900"]}')
  aws ssm send-command --region "$REGIAO" --document-name AWS-RunShellScript --instance-ids "$id" --parameters "$payload" --query Command.CommandId --output text
}
aguardar(){
  local id="$1" cmd="$2" st=''
  for _ in $(seq 1 180); do
    (( $(date +%s) < LIMITE )) || { log 'Limite de 25 minutos atingido.'; return 1; }
    st=$(aws ssm get-command-invocation --region "$REGIAO" --instance-id "$id" --command-id "$cmd" --query Status --output text 2>/dev/null || true)
    case "$st" in Success|Failed|Cancelled|TimedOut|DeliveryTimedOut|Undeliverable|Terminated) break;; esac
    sleep 5
  done
  aws ssm get-command-invocation --region "$REGIAO" --instance-id "$id" --command-id "$cmd" --query '{Estado:Status,Saida:StandardOutputContent,Erro:StandardErrorContent}' --output json || true
  [[ "$st" == Success ]]
}
rodar(){ local c; c=$(enviar "$1" "$2"); aguardar "$1" "$c"; }
substituir(){ printf '%s' "${1//__$2__/$3}"; }
LIGOU=1
log 'Iniciando EC2-01, EC2-02 e EC2-05.'
aws ec2 start-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null
aws ec2 wait instance-running --region "$REGIAO" --instance-ids "${IDS[@]}"
online=0
for _ in $(seq 1 45); do
  ativos=" $(aws ssm describe-instance-information --region "$REGIAO" --output json | jq -r '.InstanceInformationList[]|select(.PingStatus=="Online")|.InstanceId' | tr '\n' ' ') "
  online=1
  for id in "${IDS[@]}"; do [[ "$ativos" == *" $id "* ]] || online=0; done
  (( online )) && break
  sleep 10
done
(( online )) || { log 'SSM indisponivel; abortando.'; exit 1; }
for id in "${IDS[@]}"; do
  rodar "$id" $'#!/bin/bash\nset -e\nshutdown -h +25\necho PARADA_SEGURANCA_25_MIN' || exit 1
done
