Executa o teste final de escala do Flink com um e dois TaskManagers em EC2 distintas.

#!/usr/bin/env bash
set -Eeuo pipefail
export AWS_PAGER=''
REGIAO=us-east-1
BUCKET=tcc2-pacotes-CONTA_AWS-us-east-1
CONTA=CONTA_AWS
P1=i-EXEMPLO
P2=i-EXEMPLO
REMOTO=i-EXEMPLO
MOTOR=i-EXEMPLO
IP_MOTOR=IP_EXEMPLO
IP_REMOTO=IP_EXEMPLO
IDS=("$P1" "$P2" "$REMOTO" "$MOTOR")
RUN="$(date -u +%Y%m%d_%H%M%S)"
PREFIXO="resultados/escalabilidade/flink/$RUN"
LIGOU=0
INCIDENTES=0
LIMITE=$(( $(date +%s) + 2700 ))
log() { printf '[%s] %s\n' "$(date -u +%FT%TZ)" "$*"; }
finalizar() {
  local codigo=$?
  trap - EXIT INT TERM
  if (( LIGOU )); then
    log 'Desligando exclusivamente as quatro EC2 usadas nesta escala.'
    aws ec2 stop-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null || codigo=1
    if aws ec2 wait instance-stopped --region "$REGIAO" --instance-ids "${IDS[@]}"; then
      log 'Quatro EC2 confirmadas desligadas.'
    else
      log 'ATENCAO: verificar estado das quatro EC2 no console.'
      codigo=1
    fi
    log "Evidencias: s3://$BUCKET/$PREFIXO/"
  fi
  log "Codigo final: $codigo; incidentes: $INCIDENTES"
  exit "$codigo"
}
trap finalizar EXIT INT TERM

modo="${1:---check}"
[[ "$modo" = --check || "$modo" = --run ]] || {
  log 'Uso: bash tcc2_escala_flink_corrigido_20261106_v3.sh --check | --run'; exit 2;
}
for exe in aws jq; do command -v "$exe" >/dev/null || { log "Falta $exe"; exit 1; }; done
[[ "$(aws sts get-caller-identity --query Account --output text)" = "$CONTA" ]] || {
  log 'Conta AWS inesperada'; exit 1;
}
dados=$(aws ec2 describe-instances --region "$REGIAO" --instance-ids "${IDS[@]}" \
  --query 'Reservations[].Instances[].[InstanceId,State.Name,InstanceType,VpcId,IamInstanceProfile.Arn]' --output json)
[[ "$(jq length <<<"$dados")" = 4 ]] || { log 'Quatro EC2 nao localizadas'; exit 1; }
for id in "${IDS[@]}"; do
  item=$(jq -c --arg id "$id" '.[] | select(.[0]==$id)' <<<"$dados")
  [[ -n "$item" ]] || { log "Instancia desconhecida: $id"; exit 1; }
  estado=$(jq -r '.[1]' <<<"$item")
  tipo=$(jq -r '.[2]' <<<"$item")
  vpc=$(jq -r '.[3]' <<<"$item")
  iam=$(jq -r '.[4]' <<<"$item")
  log "$id: $estado | $tipo | $vpc | $iam"
  [[ "$estado" = stopped && "$tipo" = c7a.2xlarge && "$vpc" = vpc-EXEMPLO \
     && "$iam" = */TCC2-EC2-Kinesis ]] || { log 'Inventario divergente; abortado sem ligar EC2.'; exit 1; }
  pendentes=$(aws ssm list-command-invocations --region "$REGIAO" --instance-id "$id" \
    --query "CommandInvocations[?Status=='Pending'||Status=='InProgress'||Status=='Delayed'].CommandId" --output text)
  [[ -z "$pendentes" ]] || { log "SSM pendente em $id: $pendentes"; exit 1; }
done
aws s3api head-bucket --bucket "$BUCKET" >/dev/null

associacoes=$(aws ec2 describe-instances --region "$REGIAO" \
  --instance-ids "${IDS[@]}" --output json)
mapfile -t grupos_reais < <(jq -r '.Reservations[].Instances[].SecurityGroups[].GroupId' \
  <<<"$associacoes" | sort -u)
(( ${#grupos_reais[@]} > 0 )) || { log 'Nenhum grupo de seguranca encontrado.'; exit 1; }
regras=$(aws ec2 describe-security-groups --region "$REGIAO" \
  --group-ids "${grupos_reais[@]}" --output json)
if ! python3 - "$associacoes" "$regras" "$MOTOR" "$REMOTO" "$P1" "$P2" <<'PY'
import ipaddress,json,sys
inst=json.loads(sys.argv[1]); grupos=json.loads(sys.argv[2]); motor,remoto,p1,p2=sys.argv[3:]
maquinas={i['InstanceId']:i for r in inst['Reservations'] for i in r['Instances']}
regras_por_grupo={g['GroupId']:g for g in grupos['SecurityGroups']}
def liberado(origem,destino,porta):
    a,b=maquinas[origem],maquinas[destino]
    ip=ipaddress.ip_address(a['PrivateIpAddress'])
    origens={g['GroupId'] for g in a['SecurityGroups']}
    for sg in b['SecurityGroups']:
        for regra in regras_por_grupo[sg['GroupId']].get('IpPermissions',[]):
            proto=str(regra.get('IpProtocol',''))
            if not (proto=='-1' or proto=='tcp' and
                    regra.get('FromPort',0)<=porta<=regra.get('ToPort',0)):
                continue
            if any(g.get('GroupId') in origens for g in regra.get('UserIdGroupPairs',[])):
                return True
            for faixa in regra.get('IpRanges',[]):
                if ip in ipaddress.ip_network(faixa['CidrIp'],strict=False):
                    return True
    return False
def saida_liberada(origem,destino,porta):
    a,b=maquinas[origem],maquinas[destino]
    ip=ipaddress.ip_address(b['PrivateIpAddress'])
    destino_grupos={g['GroupId'] for g in b['SecurityGroups']}
    for sg in a['SecurityGroups']:
        for regra in regras_por_grupo[sg['GroupId']].get('IpPermissionsEgress',[]):
            proto=str(regra.get('IpProtocol',''))
            if not (proto=='-1' or proto=='tcp' and
                    regra.get('FromPort',0)<=porta<=regra.get('ToPort',0)):
                continue
            if any(g.get('GroupId') in destino_grupos for g in regra.get('UserIdGroupPairs',[])):
                return True
            for faixa in regra.get('IpRanges',[]):
                if ip in ipaddress.ip_network(faixa['CidrIp'],strict=False):
                    return True
    return False
requisitos=[(remoto,motor,[6123,6124,9092,5432]),
            (motor,remoto,[6122,6125]),
            (p1,motor,[9092]),(p2,motor,[9092])]
faltas=[]
for origem,destino,portas in requisitos:
    for porta in portas:
        if not liberado(origem,destino,porta):
            faltas.append((origem,destino,porta,'entrada'))
        if not saida_liberada(origem,destino,porta):
            faltas.append((origem,destino,porta,'saida'))
print('SECURITY_GROUPS_REAIS:',{i: [g['GroupId'] for g in m['SecurityGroups']]
                               for i,m in maquinas.items()})
if faltas:
    for a,b,p,tipo in faltas: print('PORTA_INTERNA_NAO_CONFIRMADA:',a,'->',b,p,tipo)
    print('ABORTADO_SEM_LIGAR_EC2. Ajustar regras de entrada ou verificar topologia.')
    sys.exit(1)
print('REGRAS_INTERNAS_CONFIRMADAS: RPC, blob, Kafka, PostgreSQL e troca de dados.')
PY
then
  log 'A comunicacao interna exigida pelo Flink nao foi comprovada. Nenhuma EC2 foi ligada.'
  exit 1
fi
log 'Inventario e regras de comunicacao interna validados sem custos EC2.'
if [[ "$modo" = --check ]]; then
  log 'CHECK OK; nenhuma instancia iniciada. --run exige autorizacao.'
  exit 0
fi
log 'ATENCAO: --run inicia QUATRO EC2 c7a.2xlarge por ate 45 minutos, alem de custos aplicaveis.'
log 'Teste planejado: duas cargas isoladas de 100.000 eventos, 1 e 2 TaskManagers em nos fisicos distintos.'
read -r -p 'Para autorizar os custos, digite SIM: ' confirmar
[[ "$confirmar" = SIM ]] || { log 'Cancelado; nenhuma EC2 iniciada.'; exit 0; }

enviar() {
  local id="$1" script="$2" parametros
  parametros=$(jq -cn --arg s "$script" '{commands:[$s],executionTimeout:["1500"]}')
  aws ssm send-command --region "$REGIAO" --document-name AWS-RunShellScript \
    --instance-ids "$id" --parameters "$parametros" --query Command.CommandId --output text
}
aguardar() {
  local id="$1" cmd="$2" status=''
  for _ in $(seq 1 300); do
    (( $(date +%s) < LIMITE )) || { log 'Limite de execucao de 45 minutos'; return 1; }
    status=$(aws ssm get-command-invocation --region "$REGIAO" --instance-id "$id" \
      --command-id "$cmd" --query Status --output text 2>/dev/null || true)
    case "$status" in
      Success|Failed|Cancelled|TimedOut|DeliveryTimedOut|Undeliverable|Terminated) break ;;
    esac
    sleep 5
  done
  aws ssm get-command-invocation --region "$REGIAO" --instance-id "$id" --command-id "$cmd" \
    --query '{Estado:Status,Saida:StandardOutputContent,Erro:StandardErrorContent}' --output json || true
  [[ "$status" = Success ]]
}
rodar() { local cmd; cmd=$(enviar "$1" "$2"); aguardar "$1" "$cmd"; }
trocar() { printf '%s' "${1//__$2__/$3}"; }

LIGOU=1
log 'Iniciando EC2-01, EC2-02, EC2-04 e EC2-05. EC2-03 permanecera desligada.'
aws ec2 start-instances --region "$REGIAO" --instance-ids "${IDS[@]}" >/dev/null
aws ec2 wait instance-running --region "$REGIAO" --instance-ids "${IDS[@]}"
online=0
for _ in $(seq 1 50); do
  lista=" $(aws ssm describe-instance-information --region "$REGIAO" \
    --query 'InstanceInformationList[?PingStatus==`Online`].InstanceId' --output text | tr '\t' ' ') "
  online=1
  for id in "${IDS[@]}"; do [[ "$lista" == *" $id "* ]] || online=0; done
  (( online )) && break
  sleep 10
done
(( online )) || { log 'SSM offline em alguma EC2'; exit 1; }
for id in "${IDS[@]}"; do
  rodar "$id" $'#!/bin/bash\nset -e\nshutdown -h +45\necho PARADA_DE_SEGURANCA_45_MIN' || exit 1
done

AUDITAR=$(cat <<'REMOTO'
#!/bin/bash
set -Eeuo pipefail
RUN=__RUN__
FASE=__FASE__
ESPERADOS=__ESPERADOS__
BASE="/tmp/tcc2-escala-flink-$RUN/$FASE"
TABELA="eventos_escala_${RUN}_${FASE}"
TOPICO="tcc2-escala-flink-$RUN-$FASE"
INICIO_EPOCA=$(date +%s)
FINAL=0
for _ in $(seq 1 180); do
  FINAL=$(docker exec postgres-resultados-tcc psql -U tcc -d tcc -t -A \
    -c "SELECT COUNT(*) FROM $TABELA" | tail -1)
  if [[ "$FINAL" = 100000 ]]; then break; fi
  sleep 5
done
FINAL_EPOCA=$(date +%s)
docker exec kafka-tcc /opt/kafka/bin/kafka-get-offsets.sh --bootstrap-server kafka:29092 \
  --topic "$TOPICO" > "$BASE/offsets_kafka.txt"
curl -fsS http://127.0.0.1:8081/taskmanagers > "$BASE/taskmanagers_depois.json"
curl -fsS http://127.0.0.1:8081/jobs/overview > "$BASE/jobs_depois.json"
export RUN FASE BASE FINAL ESPERADOS INICIO_EPOCA FINAL_EPOCA
/opt/tcc2/venv/bin/python - <<'PY'
import boto3,json,os
from pathlib import Path
base=Path(os.environ['BASE'])
managers=json.loads((base/'taskmanagers_depois.json').read_text())['taskmanagers']
jobs=json.loads((base/'jobs_depois.json').read_text())['jobs']
offsets=(base/'offsets_kafka.txt').read_text().splitlines()
disponiveis=sum(int(l.rsplit(':',1)[1]) for l in offsets if ':' in l)
info={'motor':'flink','fase':os.environ['FASE'],'taskmanagers':len(managers),
      'ids_taskmanagers':[x.get('id','') for x in managers],
      'slots_total':sum(x.get('slotsNumber',0) for x in managers),
      'esperados_eventos':100000,'offsets_kafka':disponiveis,
      'registros_sink_postgres':int(os.environ['FINAL']),
      'duracao_auditoria_segundos':int(os.environ['FINAL_EPOCA'])-int(os.environ['INICIO_EPOCA']),
      'jobs':jobs,'observacao':'JMeter mede producao; tempo de auditoria mede espera pela saida apos JMeter.'}
(base/'consistencia.json').write_text(json.dumps(info,ensure_ascii=False,indent=2))
s3=boto3.client('s3',region_name='us-east-1')
prefix=f"resultados/escalabilidade/flink/{os.environ['RUN']}/{os.environ['FASE']}/motor/"
for f in base.iterdir():
    if f.is_file() and f.name!='pipeline.sql':
        s3.upload_file(str(f),'tcc2-pacotes-CONTA_AWS-us-east-1',prefix+f.name,
                       ExtraArgs={'ServerSideEncryption':'AES256'})
print('AUDITORIA_ESCALA:',info,flush=True)
assert len(managers)==int(os.environ['ESPERADOS'])
assert disponiveis==100000 and int(os.environ['FINAL'])==100000,'Entrada/saida divergentes; S3 preservado.'
PY
REMOTO
)
